  <script>
    AuthService.preencherCabecalho();
    const estado = MscEngine.carregarEstado();
    let cacheHistorico = [];

    if (estado) {
      document.getElementById('subtituloRelatorio').textContent =
        `${estado.municipio} (${estado.codigoSiconfi}) • ${estado.periodoInicial} a ${estado.periodoFinal}`;
      renderizarTabelaMensal(estado);
    }

    document.getElementById('filtroMunicipio').addEventListener('input', renderizarHistorico);
    document.getElementById('chkSoFavoritos').addEventListener('change', renderizarHistorico);
    carregarHistorico();

    function renderizarTabelaMensal(est) {
      const tbody = document.getElementById('tbodyMensal');
      const linhas = est.meses.map(m => {
        const pct = m.rclMensal > 0 ? (m.dtpConsolidado / m.rclMensal) * 100 : 0;
        return `
          <tr>
            <td class="mono"><strong>${m.periodo}</strong></td>
            <td class="num">${MscEngine.formatarMoeda(m.recBruta)}</td>
            <td class="num">${MscEngine.formatarMoeda(m.dedRcl)}</td>
            <td class="num"><strong>${MscEngine.formatarMoeda(m.rclMensal)}</strong></td>
            <td class="num">${MscEngine.formatarMoeda(m.dtpExecutivo)}</td>
            <td class="num">${MscEngine.formatarMoeda(m.dtpLegislativo)}</td>
            <td class="num"><strong>${MscEngine.formatarPct(pct)}</strong></td>
          </tr>`;
      }).join('');

      const t = est.totais;
      tbody.innerHTML = linhas + `
        <tr style="background:#EDDCD9; font-weight:700;">
          <td>TOTAL (12M)</td>
          <td class="num">${MscEngine.formatarMoeda(est.meses.reduce((a, m) => a + m.recBruta, 0))}</td>
          <td class="num">${MscEngine.formatarMoeda(est.meses.reduce((a, m) => a + m.dedRcl, 0))}</td>
          <td class="num">${MscEngine.formatarMoeda(t.rcl)}</td>
          <td class="num">${MscEngine.formatarMoeda(t.dtpExecutivo)}</td>
          <td class="num">${MscEngine.formatarMoeda(t.dtpLegislativo)}</td>
          <td class="num">${MscEngine.formatarPct(t.pctConsolidado)}</td>
        </tr>`;
    }

    async function carregarHistorico() {
      document.getElementById('cloudLoader').style.display = 'block';
      const { data } = await CloudRepo.listarApuracoes();
      document.getElementById('cloudLoader').style.display = 'none';
      cacheHistorico = data || [];
      renderizarHistorico();
    }

    function renderizarHistorico() {
      const tbody = document.getElementById('tbodySupabase');
      const termo = document.getElementById('filtroMunicipio').value.trim().toLowerCase();
      const soFav = document.getElementById('chkSoFavoritos').checked;
      const favsLocais = JSON.parse(localStorage.getItem('sig_favs_ids') || '{}');

      const filtrados = cacheHistorico.filter(r => {
        const ehFav = Boolean(r.favorito) || Boolean(favsLocais[r.id]);
        if (soFav && !ehFav) return false;
        if (termo && !`${r.municipio} ${r.codigo_siconfi} ${r.periodo_final}`.toLowerCase().includes(termo)) return false;
        return true;
      });

      if (!filtrados.length) {
        tbody.innerHTML = `<tr><td colspan="9">Nenhum registro encontrado para o filtro atual.</td></tr>`;
        return;
      }

      tbody.innerHTML = filtrados.map(row => {
        const ehFav = Boolean(row.favorito) || Boolean(favsLocais[row.id]);
        return `
          <tr>
            <td>
              <div class="fav-wrap">
                <input type="checkbox" id="fav-${row.id}" ${ehFav ? 'checked' : ''} onchange="toggleFav(${row.id}, this.checked)" />
                <label for="fav-${row.id}" style="padding: 4px 8px; gap: 6px;">
                  <svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><path d="M20.84 4.61a5.5 5.5 0 0 0-7.78 0L12 5.67l-1.06-1.06a5.5 5.5 0 0 0-7.78 7.78l1.06 1.06L12 21.23l7.78-7.78 1.06-1.06a5.5 5.5 0 0 0 0-7.78z"></path></svg>
                  <div class="action">
                    <span class="option-1">Fav</span>
                    <span class="option-2">Sim</span>
                  </div>
                </label>
              </div>
            </td>
            <td class="mono">${new Date(row.created_at).toLocaleString('pt-BR')}</td>
            <td><strong>${row.municipio}</strong><br><small class="mono">${row.codigo_siconfi}</small></td>
            <td class="mono">${row.periodo_final}</td>
            <td class="num">${MscEngine.formatarMoeda(row.rcl)}</td>
            <td class="num">${MscEngine.formatarMoeda(row.dtp)}</td>
            <td class="num"><strong>${MscEngine.formatarPct(row.percentual_dtp)}</strong></td>
            <td style="max-width: 180px; font-size: 0.78rem;">${row.parecer_analise || '—'}</td>
            <td class="no-print">
              <button type="button" class="btn-std danger" onclick="excluirLinha(${row.id})">Excluir</button>
            </td>
          </tr>`;
      }).join('');
    }

    async function toggleFav(id, marcado) {
      await CloudRepo.alternarFavorito(id, marcado);
      const item = cacheHistorico.find(x => x.id === id);
      if (item) item.favorito = marcado;
      if (document.getElementById('chkSoFavoritos').checked) renderizarHistorico();
    }

    async function excluirLinha(id) {
      if (!confirm('Deseja realmente excluir esta análise duplicada do Supabase?')) return;
      const res = await CloudRepo.excluirApuracao(id);
      if (res.ok) {
        cacheHistorico = cacheHistorico.filter(x => x.id !== id);
        renderizarHistorico();
      } else {
        alert('Erro ao excluir: ' + (res.error?.message || 'verifique as permissões RLS.'));
      }
    }

    function exportarExcel() {
      if (!estado || !estado.meses.length) return alert('Importe os dados da MSC no Módulo 1 antes de exportar.');
      const dados = estado.meses.map(m => ({
        Codigo_Siconfi: m.codigoSiconfi, Municipio: estado.municipio, Competencia: m.periodo,
        Receitas_Correntes: Number(m.recBruta.toFixed(2)), Deducoes_RCL: Number(m.dedRcl.toFixed(2)),
        RCL_Mensal: Number(m.rclMensal.toFixed(2)), DLP_Executivo: Number(m.dtpExecutivo.toFixed(2)),
        DLP_Legislativo: Number(m.dtpLegislativo.toFixed(2)), DLP_Consolidada: Number(m.dtpConsolidado.toFixed(2))
      }));
      const ws = XLSX.utils.json_to_sheet(dados);
      const wb = XLSX.utils.book_new();
      XLSX.utils.book_append_sheet(wb, ws, 'Demonstrativo_RCL_DTP');
      XLSX.writeFile(wb, `SIG_Fiscal_${estado.codigoSiconfi}_${estado.periodoFinal}.xlsx`);
    }
