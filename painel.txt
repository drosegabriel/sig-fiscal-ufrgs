  <script>
    AuthService.preencherCabecalho();
    let estadoAtual = MscEngine.carregarEstado();
    let chartEvolucao = null, chartComposicao = null;
    const notasLocais = [];

    ['chkExec', 'chkLeg', 'chkJud'].forEach(id => {
      document.getElementById(id).addEventListener('change', () => {
        if (estadoAtual) renderizarPainel();
      });
    });

    document.getElementById('chkProvaReal').addEventListener('change', (e) => {
      document.getElementById('wrapProvaReal').style.display = e.target.checked ? 'block' : 'none';
    });

    document.getElementById('favAtual').addEventListener('change', async (e) => {
      if (!estadoAtual) return;
      const marcado = e.target.checked;
      if (estadoAtual.cloudId) {
        await CloudRepo.alternarFavorito(estadoAtual.cloudId, marcado);
      } else {
        const res = await CloudRepo.salvarApuracao(estadoAtual, notasLocais.join(' | '), marcado);
        if (res.data) { estadoAtual.cloudId = res.data.id; MscEngine.salvarEstado(estadoAtual); }
      }
    });

    if (estadoAtual) renderizarPainel();
    else document.getElementById('emptyState').style.display = 'block';

    async function carregarDemoRapido(cod) {
      estadoAtual = await MscEngine.carregarBaseDemo(cod, false);
      document.getElementById('emptyState').style.display = 'none';
      renderizarPainel();
    }

    async function adicionarNota() {
      const inp = document.getElementById('inputNota');
      const txt = inp.value.trim();
      if (!txt || !estadoAtual) return;
      notasLocais.push(txt);
      inp.value = '';

      const ul = document.getElementById('listaNotas');
      ul.innerHTML = notasLocais.map(n => `<li><strong>Parecer:</strong> ${n}</li>`).join('');

      const fav = document.getElementById('favAtual').checked;
      const res = await CloudRepo.salvarApuracao(estadoAtual, notasLocais.join(' | '), fav);
      if (res.data) { estadoAtual.cloudId = res.data.id; MscEngine.salvarEstado(estadoAtual); }
    }

    function renderizarPainel() {
      document.getElementById('dashboardContent').style.display = 'block';
      document.getElementById('favBoxPainel').style.display = 'inline-block';
      document.getElementById('tituloMunicipio').textContent = `${estadoAtual.municipio} (${estadoAtual.codigoSiconfi})`;
      document.getElementById('subtituloPeriodo').textContent =
        `Janela de 12 meses: ${estadoAtual.periodoInicial} a ${estadoAtual.periodoFinal} (MSC Agregada — period_change)`;

      const incExec = document.getElementById('chkExec').checked;
      const incLeg = document.getElementById('chkLeg').checked;
      const incJud = document.getElementById('chkJud').checked;

      const t = estadoAtual.totais;
      document.getElementById('kpiRcl').textContent = MscEngine.formatarMoeda(t.rcl);
      document.getElementById('kpiRclAjustada').textContent = MscEngine.formatarMoeda(t.rclAjustada);
      document.getElementById('lblDedAjuste').textContent = t.dedAjuste > 0
        ? `Deduções emendas/ACS: ${MscEngine.formatarMoeda(t.dedAjuste)}`
        : 'Base legal única do Município (RREO Anexo 3)';

      let valorDtp = (incExec ? t.dtpExecutivo : 0) + (incLeg ? t.dtpLegislativo : 0) + (incJud ? t.dtpJudiciario : 0);
      let limiteMax = (incExec ? 54 : 0) + (incLeg ? 6 : 0);
      if (limiteMax === 0) limiteMax = 60;

      const pctDtp = t.rclAjustada > 0 ? (valorDtp / t.rclAjustada) * 100 : 0;
      const nomesPoderes = [];
      if (incExec) nomesPoderes.push('Executivo');
      if (incLeg) nomesPoderes.push('Legislativo');
      if (incJud) nomesPoderes.push('Judiciário');

      document.getElementById('lblDtpTitulo').textContent = `DLP — ${nomesPoderes.join(' + ') || 'Nenhum'}`;
      document.getElementById('lblDtpNota').textContent = incJud && !incExec && !incLeg
        ? 'Municípios não possuem Poder Judiciário (art. 20 LRF)'
        : 'Despesas executadas/liquidadas (sem Restos a Pagar)';
      document.getElementById('kpiDtp').textContent = MscEngine.formatarMoeda(valorDtp);
      document.getElementById('kpiPct').textContent = MscEngine.formatarPct(pctDtp);

      const badge = document.getElementById('badgeLimite');
      if (pctDtp >= limiteMax) badge.innerHTML = `<span class="badge badge-danger">Acima do Máximo (${limiteMax}%)</span>`;
      else if (pctDtp >= limiteMax * 0.95) badge.innerHTML = `<span class="badge badge-warn">Prudencial (&ge; ${(limiteMax*0.95).toFixed(1)}%)</span>`;
      else if (pctDtp >= limiteMax * 0.90) badge.innerHTML = `<span class="badge badge-warn">Alerta (&ge; ${(limiteMax*0.90).toFixed(1)}%)</span>`;
      else badge.innerHTML = `<span class="badge badge-ok">Regular (&lt; ${(limiteMax*0.90).toFixed(1)}%)</span>`;

      document.getElementById('tbodyPoderes').innerHTML = `
        <tr><td><strong>Poder Executivo</strong></td><td class="mono">10131 / 10132</td><td class="num">${MscEngine.formatarMoeda(t.dtpExecutivo)}</td><td class="num">${MscEngine.formatarPct(t.pctExecutivo)}</td><td class="num">54,00%</td></tr>
        <tr><td><strong>Poder Legislativo</strong></td><td class="mono">20231</td><td class="num">${MscEngine.formatarMoeda(t.dtpLegislativo)}</td><td class="num">${MscEngine.formatarPct(t.pctLegislativo)}</td><td class="num">6,00%</td></tr>
        <tr><td><strong>Poder Judiciário</strong></td><td class="mono">303XX (N/A Mun.)</td><td class="num">${MscEngine.formatarMoeda(t.dtpJudiciario)}</td><td class="num">${MscEngine.formatarPct(t.pctJudiciario)}</td><td class="num">N/A</td></tr>
        <tr style="background:#EDDCD9; font-weight:700;"><td>Total Consolidado</td><td class="mono">Município</td><td class="num">${MscEngine.formatarMoeda(t.dtpConsolidado)}</td><td class="num">${MscEngine.formatarPct(t.pctConsolidado)}</td><td class="num">60,00%</td></tr>
      `;

      renderizarGraficos(incExec, incLeg, incJud);
      renderizarProvaReal();
    }

    function renderizarGraficos(incExec, incLeg, incJud) {
      const labels = estadoAtual.meses.map(m => m.periodo);
      const dadosRcl = estadoAtual.meses.map(m => m.rclMensal);
      const dadosDtp = estadoAtual.meses.map(m =>
        (incExec ? m.dtpExecutivo : 0) + (incLeg ? m.dtpLegislativo : 0) + (incJud ? (m.dtpJudiciario || 0) : 0)
      );
      const dadosPct = estadoAtual.meses.map((m, i) => m.rclMensal > 0 ? Number(((dadosDtp[i] / m.rclMensal) * 100).toFixed(2)) : 0);

      if (chartEvolucao) chartEvolucao.destroy();
      chartEvolucao = new Chart(document.getElementById('chartEvolucao'), {
        data: {
          labels,
          datasets: [
            { type: 'bar', label: 'RCL Mensal (R$)', data: dadosRcl, backgroundColor: '#264143', borderRadius: 4, yAxisID: 'y' },
            { type: 'bar', label: 'DLP Selecionada (R$)', data: dadosDtp, backgroundColor: '#DE5499', borderRadius: 4, yAxisID: 'y' },
            { type: 'line', label: '% DLP / RCL (Mês)', data: dadosPct, borderColor: '#E99F4C', backgroundColor: '#E99F4C', borderWidth: 3, tension: 0.3, yAxisID: 'y1' }
          ]
        },
        options: {
          responsive: true, maintainAspectRatio: false,
          scales: {
            y: { position: 'left', beginAtZero: true },
            y1: { position: 'right', beginAtZero: true, grid: { drawOnChartArea: false }, ticks: { callback: v => v + '%' } }
          }
        }
      });

      // GRÁFICO DE ROSCA DINÂMICO POR PODER (com R$ e % no tooltip)
      let pieLabels, pieData, pieColors;
      if (incLeg && !incExec) {
        document.getElementById('tituloPizza').textContent = 'Participação do Legislativo na DTP Total (12M)';
        document.getElementById('subPizza').textContent = 'Câmara Municipal (PO 20231) frente ao restante da folha municipal.';
        pieLabels = ['Poder Legislativo (PO 20231)', 'Demais Poderes (Executivo)'];
        pieData = [estadoAtual.totais.dtpLegislativo, estadoAtual.totais.dtpExecutivo];
        pieColors = ['#DE5499', '#cbd5e1'];
      } else if (incExec && !incLeg) {
        document.getElementById('tituloPizza').textContent = 'Executivo: Folha vs. Margem Livre da RCL (12M)';
        document.getElementById('subPizza').textContent = 'Parcela da RCL Ajustada comprometida pelo Poder Executivo.';
        const livre = Math.max(0, estadoAtual.totais.rclAjustada - estadoAtual.totais.dtpExecutivo);
        pieLabels = ['DLP Poder Executivo', 'Demais Recursos da RCL'];
        pieData = [estadoAtual.totais.dtpExecutivo, livre];
        pieColors = ['#264143', '#E99F4C'];
      } else {
        document.getElementById('tituloPizza').textContent = 'Composição das Receitas Correntes (12M)';
        document.getElementById('subPizza').textContent = 'Distribuição oficial por Natureza da Receita (NR 11, NR 17 e Outras).';
        const c = estadoAtual.composicaoReceita;
        pieLabels = ['Impostos e Taxas (NR 11)', 'Transferências Correntes (NR 17)', 'Outras Receitas Correntes'];
        pieData = [c.impostos, c.transferencias, c.outras];
        pieColors = ['#264143', '#DE5499', '#E99F4C'];
      }

      const somaPie = pieData.reduce((a, b) => a + Math.abs(b), 0) || 1;
      if (chartComposicao) chartComposicao.destroy();
      chartComposicao = new Chart(document.getElementById('chartComposicao'), {
        type: 'doughnut',
        data: { labels: pieLabels, datasets: [{ data: pieData, backgroundColor: pieColors, borderColor: '#fff', borderWidth: 3 }] },
        options: {
          responsive: true, maintainAspectRatio: false,
          layout: { padding: { bottom: 12 } },
          plugins: {
            legend: { position: 'bottom', labels: { boxWidth: 14, padding: 14 } },
            tooltip: {
              callbacks: {
                label: (ctx) => {
                  const val = ctx.parsed || 0;
                  const pct = ((val / somaPie) * 100).toFixed(1).replace('.', ',');
                  return ` ${ctx.label}: ${MscEngine.formatarMoeda(val)} (${pct}%)`;
                }
              }
            }
          }
        }
      });
    }

    function renderizarProvaReal() {
      const tbody = document.getElementById('tbodyProvaReal');
      const ref = window.OFICIAL_SICONFI[estadoAtual.codigoSiconfi];
      if (!ref) {
        tbody.innerHTML = `<tr><td colspan="4">Município (${estadoAtual.codigoSiconfi}) sem gabarito pré-cadastrado. Apuração 100% dinâmica pela MSC.</td></tr>`;
        return;
      }
      tbody.innerHTML = `
        <tr><td>RCL (12M)</td><td class="num">${MscEngine.formatarMoeda(estadoAtual.totais.rcl)}</td><td class="num">${MscEngine.formatarMoeda(ref.rcl)}</td><td class="num">${MscEngine.formatarMoeda(estadoAtual.totais.rcl - ref.rcl)}</td></tr>
        <tr><td>DLP Executivo</td><td class="num">${MscEngine.formatarMoeda(estadoAtual.totais.dtpExecutivo)}</td><td class="num">${MscEngine.formatarMoeda(ref.dtpExec)}</td><td class="num">${MscEngine.formatarMoeda(estadoAtual.totais.dtpExecutivo - ref.dtpExec)}</td></tr>
      `;
    }
